	public string AuthorizationHeaderText
	{
		get
		{
			string text = "Unknown:";
			switch (Main.Master.ConnectionInfo.ConnectUserType)
			{
			case LM.Auth.Models.UserType.NormalUser:
				text = "Basic:";
				break;
			case LM.Auth.Models.UserType.WebUser:
				text = "Web:";
				break;
			case LM.Auth.Models.UserType.ServiceUser:
				text = "Service:";
				break;
			}
			return text + EscapeColonDelimiter(Main.Master.ConnectionInfo.ConnectUserLogin + "@" + Main.Master.ConnectionInfo.ConnectAgency) + ":" + EscapeColonDelimiter(Main.Master.ConnectionInfo.ConnectPassword) + ":" + EscapeColonDelimiter(Main.Master.ConnectionInfo.ConnectionID.ToString("X")) + ":" + EscapeColonDelimiter(SourceAddress) + ":MinuteTraq:" + EscapeColonDelimiter(ProgramVersion.ToString(4)) + ":" + EscapeColonDelimiter(ExUtil.GetDotNetFWInfo());
		}
	}

	public static DataSet GetChanges(DB ds, params DataRow[] oExtraRows)
	{
		ds = (DB)ds.Copy();
		ds.Relations.Clear();
		ds = (DB)ds.GetChanges();
		foreach (DataTable table in ds.Tables)
		{
			foreach (DataRow row2 in table.Rows)
			{
				if (row2.RowState == DataRowState.Modified && !HasChanges(row2))
				{
					row2.RejectChanges();
				}
			}
		}
		DB dB = new DB();
		DB dB2 = (DB)ds.GetChanges();
		if (dB2 == null)
		{
			return null;
		}
		dB.Merge(dB2);
		foreach (DataRow row in oExtraRows)
		{
			MergeRow(dB, row);
		}
		return dB;
	}
